8) DELETE ALL FIREBASE AUTH USERS (IRREVERSIBLE)
==================================================
cd /Users/sujeetkumarsingh/Desktop/MuhDikhai/PlasticWorld
set -a
source .env
set +a

node <<'NODE'
const admin = require('firebase-admin');

const projectId = process.env.FIREBASE_PROJECT_ID;
const clientEmail = process.env.FIREBASE_CLIENT_EMAIL;
const privateKey = (process.env.FIREBASE_PRIVATE_KEY || '').replace(/\\n/g, '\n');

if (!projectId || !clientEmail || !privateKey) {
  console.error('Missing Firebase env vars. Check .env');
  process.exit(1);
}

admin.initializeApp({
  credential: admin.credential.cert({ projectId, clientEmail, privateKey }),
});

async function deleteAllUsers() {
  let nextPageToken = undefined;
  let totalDeleted = 0;

  do {
    const listResult = await admin.auth().listUsers(1000, nextPageToken);
    const uids = listResult.users.map((u) => u.uid);
    if (uids.length > 0) {
      const result = await admin.auth().deleteUsers(uids);
      totalDeleted += result.successCount;
      if (result.failureCount > 0) {
        console.error(`Batch failures: ${result.failureCount}`);
      }
    }
    nextPageToken = listResult.pageToken;
  } while (nextPageToken);

  console.log(`Deleted Firebase Auth users: ${totalDeleted}`);
}

deleteAllUsers()
  .then(() => process.exit(0))
  .catch((err) => {
    console.error(err);
    process.exit(1);
  });
NODE
